# Minimal requirements for exporting a CBCT->CT model to ONNX and serving it
# in LaTIM-SNAP (latimsnap-i2i). No MONAI/generative/kornia needed.
#
# torch: choose the wheel matching your machine (CPU vs CUDA). For a CPU-only
# export/serving box:
#   uv pip install torch --index-url https://download.pytorch.org/whl/cpu
# On a GPU machine, drop `--index-url` and uv picks the CUDA build (which
# bundles the CUDA/cuDNN runtimes, so no system toolkit is required).
torch>=2.1
onnx>=1.14
onnxscript>=0.1
onnxruntime>=1.14
# For GPU inference install onnxruntime-gpu (same version) instead of the CPU
# onnxruntime above; the server auto-uses the CUDAExecutionProvider and
# auto-exposes the bundled NVIDIA libs via LD_LIBRARY_PATH.
#   uv pip install onnxruntime-gpu
numpy>=1.24
nibabel>=3.0
